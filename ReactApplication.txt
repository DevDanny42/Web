App.jsx
import { useState, type FormEvent, type ChangeEvent } from "react";
import {
Routes,
Route,
Link
} from "react-router-dom";
function Navbar() {
return (
<nav className="navbar">
<h2>Student SPA</h2>
<div>
<Link to="/">Home</Link>
<Link to="/students">Students</Link>
<Link to="/about">About</Link>
</div>
</nav>

);
}
function Home() {
return (
<div className="container">
<h1>Welcome toStudent Management System</h1>
<p>
This is a Single Page Application developed using Advanced React.
</p>
<Link className="button" to="/students">
View Students
</Link>
</div>
);
}

function Students() {
const [students, setStudents] = useState([
{ id: 1, name: "Rahul", course: "Computer Science" },
{ id: 2, name: "Priya", course: "Information Technology" }

]);
const [name, setName] = useState("");
const [course, setCourse] = useState("");
// Form event type
const addStudent = (e: FormEvent<HTMLFormElement>) => {
e.preventDefault();
if (name.trim() === "" || course.trim() === "") {
alert("Please enter all details");
return;
}
const newStudent= {
id: Date.now(),
name: name,
course: course
};

setStudents([...students, newStudent]);
setName("");
setCourse("");
};

// Name input event
const handleNameChange = (e: ChangeEvent<HTMLInputElement>) => {
setName(e.target.value);
};
// Course input event
const handleCourseChange = (e: ChangeEvent<HTMLInputElement>) => {
setCourse(e.target.value);
};

return (
<div className="container">
<h1>Student Management</h1>
<form onSubmit={addStudent} className="form">
<input
type="text"
placeholder="Enter student name"
value={name}
onChange={handleNameChange}
/>
<input

type="text"
placeholder="Enter course"
value={course}
onChange={handleCourseChange}
/>

<button type="submit">
Add Student
</button>
</form>
<h2>Student List</h2>
{students.length === 0 ? (
<p>No students available.</p>
) : (
<div>
{students.map((student) => (
<div className="student-card" key={student.id}>
<h3>{student.name}</h3>
<p>Course: {student.course}</p>
</div>

))}
</div>
)}
</div>
);
}

function About() {
return (
<div className="container">
<h1>About</h1>
<p>
This Single Page Application demonstrates advanced React
concepts including components, state, event handling and
client-side routing.
</p>
</div>
);
}

function App() {
return (
<>
<Navbar />

<Routes>
<Route path="/" element={<Home />} />
<Route path="/students" element={<Students />} />
<Route path="/about" element={<About />} />
</Routes>
</>
);
}
export default App;
