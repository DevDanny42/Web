import React from 'react'
import { useState } from 'react';
function Welcome(prop){
  return (
    <div>
      <h1>Welcome {prop.name}</h1>
    </div>
  );
}


function Counter(){
  const [count,setCount] =useState(0);
  return (
    <div>
      <h1>{count}</h1>
      <button onClick={()=>setCount(count+1)}>Click me ! </button>
    </div>
  );

}

function App() {
  return (
    <div>
      <p>hello</p>
    <Welcome name="danny"/>
    <Counter></Counter>
    </div>
  )
}

export default App;
